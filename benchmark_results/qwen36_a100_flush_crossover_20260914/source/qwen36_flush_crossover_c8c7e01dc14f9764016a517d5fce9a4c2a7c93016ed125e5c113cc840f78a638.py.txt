"""Controlled history-distance and flush-policy measurements for Qwen GDN."""

import argparse
import gc
import json
import random
import statistics
from pathlib import Path

import torch

from vllm.model_executor.layers.fla.ops.fused_sigmoid_gating import (
    fused_sigmoid_gating_delta_rule_update,
)
from vllm.model_executor.layers.fla.ops.gdn_replayssm_spec_decode import (
    commit_gdn_replayssm_spec,
    gdn_replayssm_spec_decode,
)


def save(path, value):
    path.parent.mkdir(parents=True, exist_ok=True)
    tmp = path.with_suffix(".tmp")
    tmp.write_text(json.dumps(value, indent=2) + "\n")
    tmp.replace(path)


class Inputs:
    def __init__(self, batch, draft, seed):
        torch.manual_seed(seed)
        self.batch, self.width = batch, draft + 1
        self.a_log = torch.randn(32, device="cuda")
        self.bias = torch.randn(32, device="cuda")
        self.qkv = torch.randn(
            batch * self.width, 8192, device="cuda", dtype=torch.bfloat16
        )
        self.a = torch.randn(
            batch * self.width, 32, device="cuda", dtype=torch.bfloat16
        )
        self.b = torch.randn_like(self.a)
        self.q, self.k, self.v = [
            x.contiguous().reshape(1, batch * self.width, heads, 128)
            for x, heads in zip(self.qkv.split([2048, 2048, 4096], -1), [16, 16, 32])
        ]
        self.slots = torch.arange(1, batch + 1, device="cuda", dtype=torch.int32)
        self.qsl = torch.arange(batch + 1, device="cuda", dtype=torch.int32)
        self.qsl *= self.width
        self.indices = torch.arange(
            1, batch * self.width + 1, device="cuda", dtype=torch.int32
        ).reshape(batch, self.width)
        self.accepted = torch.ones(batch, device="cuda", dtype=torch.int32)
        self.initial = torch.randn(batch + 1, 32, 128, 128, device="cuda") * 0.1
        self.states = [self.initial.clone()]
        self.history_d = torch.zeros(
            batch + 1, 32, 128, 128, device="cuda", dtype=torch.float16
        )
        self.history_k = torch.zeros(
            batch + 1, 16, 128, 128, device="cuda", dtype=torch.float16
        )
        self.history_g = torch.zeros(batch + 1, 32, 128, device="cuda")
        state = self.initial[1:].clone()
        for h in range(64):
            key = torch.randn(batch, 16, 128, device="cuda", dtype=torch.bfloat16)
            key = torch.nn.functional.normalize(key.float(), dim=-1, eps=1e-6)
            value = torch.randn(batch, 32, 128, device="cuda", dtype=torch.bfloat16)
            a = torch.randn(batch, 32, device="cuda", dtype=torch.bfloat16)
            beta = torch.randn_like(a).float().sigmoid()
            g = -self.a_log.exp() * torch.nn.functional.softplus(a.float() + self.bias)
            expanded = key.repeat_interleave(2, dim=1)
            state = state * g.exp()[..., None, None]
            delta = beta[..., None] * (
                value.float() - (state * expanded[..., None, :]).sum(-1)
            )
            state = state + delta[..., :, None] * expanded[..., None, :]
            self.history_d[1:, :, h] = delta.half()
            self.history_k[1:, :, h] = key.half()
            self.history_g[1:, :, h] = g
            snapshot = self.initial.clone()
            snapshot[1:] = state
            self.states.append(snapshot)


class Layer:
    def __init__(self, inputs):
        self.x = inputs
        b, t = inputs.batch, inputs.width
        self.checkpoint = inputs.initial.clone()
        self.d = inputs.history_d.clone()
        self.k = inputs.history_k.clone()
        self.g = inputs.history_g.clone()
        self.state = torch.zeros(b * t + 1, 32, 128, 128, device="cuda")
        self.out = torch.empty(b * t, 32, 128, device="cuda", dtype=torch.bfloat16)

    def reset(self, history):
        self.checkpoint.copy_(self.x.initial)
        self.d.copy_(self.x.history_d)
        self.k.copy_(self.x.history_k)
        self.g.copy_(self.x.history_g)
        self.state[self.x.indices[:, 0].long()] = self.x.states[history][1:]

    def standard(self):
        x = self.x
        return fused_sigmoid_gating_delta_rule_update(
            A_log=x.a_log,
            a=x.a,
            b=x.b,
            dt_bias=x.bias,
            q=x.q,
            k=x.k,
            v=x.v,
            initial_state=self.state,
            cu_seqlens=x.qsl,
            ssm_state_indices=x.indices,
            num_accepted_tokens=x.accepted,
            use_qk_l2norm_in_kernel=True,
        )[0]

    def replay(self, wp, base, flags):
        x = self.x
        gdn_replayssm_spec_decode(
            mixed_qkv=x.qkv,
            a=x.a,
            b=x.b,
            A_log=x.a_log,
            dt_bias=x.bias,
            checkpoint_state=self.checkpoint,
            d_cache=self.d,
            k_cache=self.k,
            g_cache=self.g,
            out=self.out,
            query_start_loc=x.qsl,
            ssm_state_indices=x.slots,
            write_pos=wp,
            cache_base=base,
            is_flush=flags,
            max_cache_len=64 + x.width,
            max_spec_len=x.width,
        )
        return self.out


def cursors(batch, history=0, flush=False):
    wp = torch.full((batch + 1,), history, device="cuda", dtype=torch.int32)
    base = torch.zeros_like(wp)
    flags = torch.full((batch + 1,), int(flush), device="cuda", dtype=torch.int8)
    return wp, base, flags


def capture(fn):
    for _ in range(3):
        fn()
    torch.accelerator.synchronize()
    graph = torch.cuda.CUDAGraph()
    with torch.cuda.graph(graph):
        fn()
    return graph


def time_graph(graph):
    # Queue work ahead of the events so CPU submission gaps are not timed.
    if not hasattr(time_graph, "prelude"):
        a = torch.ones(1024, 1024, device="cuda")
        out = torch.empty_like(a)

        def delay():
            for _ in range(8):
                torch.mm(a, a, out=out)

        time_graph.prelude = capture(delay)
        time_graph.buffers = (a, out)
    start = torch.Event(device="cuda", enable_timing=True)
    end = torch.Event(device="cuda", enable_timing=True)
    time_graph.prelude.replay()
    start.record()
    graph.replay()
    end.record()
    end.synchronize()
    return start.elapsed_time(end) * 1000


def distance(args, inputs, layers):
    b, t = inputs.batch, inputs.width
    cursor = cursors(b)
    rows = []
    outpath = Path(args.output) / f"distance_b{b}_d{t - 1}_s{args.seed}.json"
    if outpath.exists():
        rows = json.loads(outpath.read_text())
    done = {(r["history"], r["layers"]) for r in rows}
    for count in (1, len(layers)):
        active = layers[:count]

        def standard(active=active):
            for layer in active:
                layer.standard()

        def replay(active=active):
            for layer in active:
                layer.replay(*cursor)

        graphs = {"standard": capture(standard), "replay": capture(replay)}
        for h in range(65):
            if (h, count) in done:
                continue
            values = {name: [] for name in ("standard", "replay", "flush")}
            errors = {}
            for layer in active:
                layer.reset(h)
            reference = active[0].standard().reshape_as(active[0].out).clone()
            for name in ("replay", "flush"):
                active[0].reset(h)
                cursor[0].fill_(h)
                cursor[1].zero_()
                cursor[2].fill_(int(name == "flush"))
                actual = active[0].replay(*cursor)
                torch.testing.assert_close(actual, reference, rtol=0.04, atol=0.01)
                errors[name] = (actual.float() - reference.float()).abs().max().item()
            for repeat in range(args.repeats):
                names = list(values)
                random.Random(1000 * h + repeat).shuffle(names)
                for name in names:
                    for layer in active:
                        layer.reset(h)
                    cursor[0].fill_(h)
                    cursor[1].zero_()
                    cursor[2].fill_(int(name == "flush"))
                    graph = graphs["standard" if name == "standard" else "replay"]
                    values[name].append(time_graph(graph) / count)
            row = dict(
                batch=b,
                draft=t - 1,
                history=h,
                layers=count,
                seed=args.seed,
                us=values,
                max_abs_error=errors,
                default_nonflush=h < 65 - t,
                correctness=True,
            )
            rows.append(row)
            save(outpath, rows)
            print(
                "DIST",
                b,
                t - 1,
                h,
                count,
                {k: round(statistics.median(v), 3) for k, v in values.items()},
                flush=True,
            )
        del graphs


def trace_acceptance(path, batch, width, steps):
    rows = json.loads(Path(path).read_text())
    first_cursor = rows[0]["cursor_id"]
    rows = [row for row in rows if row["cursor_id"] == first_cursor]
    tracks = {}
    for row in rows:
        ids = row["request_ids"]
        if len(ids) != len(row["accepted_previous"]):
            continue
        for request, accepted in zip(ids, row["accepted_previous"]):
            if request is not None and 1 <= accepted <= width:
                tracks.setdefault(request, []).append(accepted)
    tracks = [x[1:] for x in tracks.values() if len(x) > 1]
    if not tracks:
        raise ValueError("No usable per-request acceptance trace")
    return [
        [
            tracks[i % len(tracks)][s % len(tracks[i % len(tracks)])]
            for i in range(batch)
        ]
        for s in range(steps)
    ]


def policy(args, inputs, layers):
    b, t = inputs.batch, inputs.width
    path = Path(args.output) / f"policy_b{b}_d{t - 1}_s{args.seed}.json"
    rows = json.loads(path.read_text()) if path.exists() else []
    done = {(r["track"], r["interval"]) for r in rows}
    steps = args.steps
    tracks = {"natural": trace_acceptance(args.trace, b, t, steps)}
    tracks.update(
        {
            f"fixed_{a}": [[a] * b for _ in range(steps)]
            for a in sorted({1, min(4, t), t})
        }
    )
    # The overflow guard already flushes at history >= 65 - T.
    intervals = [None, *range(1, 65 - t)]
    cursor = cursors(b)
    for name, track in tracks.items():
        accepts = [torch.tensor(a, device="cuda", dtype=torch.int32) for a in track]
        tokens = sum(map(sum, track))
        for interval in intervals:
            if (name, interval) in done:
                continue

            def cycle(accepts=accepts, interval=interval):
                for accepted in accepts:
                    for layer in layers:
                        layer.replay(*cursor)
                    commit_gdn_replayssm_spec(
                        *cursor,
                        accepted,
                        inputs.slots,
                        64 + t,
                        t,
                        cache_buf_len=128,
                        flush_interval=interval,
                    )

            graph = capture(cycle)
            values = []
            for _ in range(args.repeats):
                for layer in layers:
                    layer.reset(0)
                cursor[0].zero_()
                cursor[1].zero_()
                cursor[2].zero_()
                values.append(time_graph(graph) / tokens)
            row = dict(
                batch=b,
                draft=t - 1,
                seed=args.seed,
                track=name,
                interval=interval,
                layers=len(layers),
                steps=steps,
                committed_tokens=tokens,
                us_per_token=values,
            )
            rows.append(row)
            save(path, rows)
            print(
                "POLICY",
                b,
                t - 1,
                name,
                interval,
                round(statistics.median(values), 3),
                flush=True,
            )
            del graph


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--output", required=True)
    parser.add_argument("--batch", type=int, required=True)
    parser.add_argument("--draft", type=int, required=True)
    parser.add_argument("--seed", type=int, default=0)
    parser.add_argument("--layers", type=int, default=30)
    parser.add_argument("--repeats", type=int, default=21)
    parser.add_argument("--steps", type=int, default=128)
    parser.add_argument("--mode", choices=["distance", "policy"], required=True)
    parser.add_argument("--trace")
    parser.add_argument("--completion")
    args = parser.parse_args()
    if args.mode == "policy" and not args.trace:
        parser.error("--trace is required for policy mode")
    inputs = Inputs(args.batch, args.draft, args.seed)
    layers = [Layer(inputs) for _ in range(args.layers)]
    if args.mode == "distance":
        distance(args, inputs, layers)
    else:
        policy(args, inputs, layers)
    del layers, inputs
    gc.collect()
    if args.completion:
        save(Path(args.completion), {"complete": True, "arguments": vars(args)})


if __name__ == "__main__":
    main()
